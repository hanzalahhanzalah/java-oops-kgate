import csv
record=int(input("how many books you want to add; "))
with open("book.csv","w",newline="") as x:

    file=csv.writer(x)
    for _ in range(record):
        file.writerow([input("enter title"),input("enter author")])
search=input("enter author name to display")
foundbooks=[]
with open("book.csv","r") as x:
    reader=csv.reader(x)
    foundbooks=[b[0] for b in reader if b[1]==search]
if foundbooks:
    print(f"book written by{search}")
    print(*foundbooks, sep="\n")
else:
    print(f"no book written by{search}")
